_LOCAL_HASH_NAV_FIX = """<script data-idealab-file-nav>
/* Keep in-page links working inside file:// and unique-origin preview frames. */
document.addEventListener("click",function(event){
  var link=event.target.closest&&event.target.closest('a[href^="#"]');
  if(!link)return;
  var hash=link.getAttribute("href");
  if(!hash||hash==="#")return;
  var id;
  try{id=decodeURIComponent(hash.slice(1));}catch(_){id=hash.slice(1);}
  var target=document.getElementById(id);
  if(!target)return;
  event.preventDefault();
  target.scrollIntoView({behavior:"smooth",block:"start"});
},{capture:true});
</script>"""


def make_preview_file_safe(document: str) -> str:
    """Harden a standalone preview for local file/unique-origin navigation."""
    html = document or ""
    if "data-idealab-file-nav" in html:
        return html
    if not re.search(r"""<a\b[^>]*\bhref\s*=\s*["']#[^"']+["']""", html, re.I):
        return html
    close = re.search(r"</body\s*>", html, re.I)
    if close:
        return html[: close.start()] + _LOCAL_HASH_NAV_FIX + "\n" + html[close.start() :]
    return html + "\n" + _LOCAL_HASH_NAV_FIX


def module_list(user=None) -> list[dict]:
    import curriculum

    owned = curriculum.unlocked_set(user)
    rows = []
    for module in MODULES.values():
        if module.get("listed") is False or module["id"] in {"idea-watch", "game-port"}:
            continue
        item = dict(module)
        needed = curriculum.required_skills(module["id"])
        ok, reason = curriculum.can_use_module(user, module["id"])
        item["locked"] = not ok
        item["lock_reason"] = "" if ok else reason
        item["required_skills"] = needed
        item["unlocked_skills"] = [name for name in needed if name in owned]
        item["quiz_skill"] = next((name for name in needed if name in curriculum.QUIZZES), "")
        if module["id"] == "style-image":
            item["task_specs"] = [
                task_specs.get_spec("style-image", "spdesign"),
                task_specs.get_spec("style-image", "dtdesign"),
            ]
        else:
            item["task_specs"] = [task_specs.get_spec(module["id"])]
        rows.append(item)
    for product in PLAY_PRODUCTS.values():
        item = dict(product)
        item["locked"] = False
        item["lock_reason"] = ""
        item["required_skills"] = []
        item["unlocked_skills"] = []
        item["quiz_skill"] = ""
        item["task_specs"] = []
        rows.append(item)
    return rows


def _schedule_hours(value) -> int:
    try:
        hours = int(value)
    except (TypeError, ValueError):
        hours = 6
    return hours if hours in {1, 6, 12, 24} else 6


def _new_id() -> str:
    return secrets.token_urlsafe(8)[:11]


def _safe_name(name: str) -> str:
    stem = Path(name or "asset").name
    stem = re.sub(r"[^A-Za-z0-9._\-\u4e00-\u9fff]+", "_", stem)
    return stem[:120] or "asset"


def _project_dir(project_id: str) -> Path:
    if not PROJECT_ID_RE.match(project_id or ""):
        raise StudioError("项目 ID 非法")
    return PROJECTS_DIR / project_id


def user_inputs_snapshot(project: dict) -> dict:
    """Freeze the user's prompt and other creation inputs for replay / remake."""
    assets = []
    for asset in project.get("assets") or []:
        assets.append(
            {
                "name": asset.get("name") or "",
                "original_name": asset.get("original_name") or asset.get("name") or "",
                "content_type": asset.get("content_type") or "",
                "size": int(asset.get("size") or 0),
                "source_url": asset.get("source_url") or "",
                "source_type": asset.get("source_type") or "",
            }
        )
    return {
        "module": project.get("module") or "",
        "title": project.get("title") or "",
        "prompt": project.get("brief") or "",
        "brief": project.get("brief") or "",
        "track": project.get("track") or "",
        "animations": project.get("animations") or "",
        "character_type": project.get("character_type") or "",
        "anatomy_confirmed": bool(project.get("anatomy_confirmed")),
        "scene_scope": project.get("scene_scope") or "",
        "core_interaction": project.get("core_interaction") or "",
        "schedule_hours": int(project.get("schedule_hours") or 0),
        "game_slug": project.get("game_slug") or "",
        "assets": assets,
        "asset_count": len(assets),
        "captured_at": time.time(),
    }


def _inputs_markdown(project: dict) -> str:
    inputs = project.get("user_inputs") or user_inputs_snapshot(project)
    lines = [
        f"# 用户输入 · {inputs.get('title') or project.get('id')}",
        "",
        f"- 模块：`{inputs.get('module') or ''}`",
        f"- 轨道：`{inputs.get('track') or '—'}`",
        "",
        "## 提示词 / 创作描述",
        "",
        inputs.get("prompt") or inputs.get("brief") or "（空）",
        "",
    ]
    if inputs.get("animations"):
        lines.extend(["## 动作列表", "", str(inputs["animations"]), ""])
    if inputs.get("character_type"):
        lines.append(f"- 角色类型：{inputs['character_type']}")
        lines.append(f"- 解剖确认：{'是' if inputs.get('anatomy_confirmed') else '否'}")
        lines.append("")
    if inputs.get("scene_scope") or inputs.get("core_interaction"):
        lines.extend(
            [
                "## 游戏设定",
                "",
                f"- 场景范围：{inputs.get('scene_scope') or '—'}",
                f"- 核心交互：{inputs.get('core_interaction') or '—'}",
                "",
            ]
        )
    lines.extend(["## 素材", ""])
    assets = inputs.get("assets") or []
    if not assets:
        lines.append("（无）")
    else:
        for asset in assets:
            note = asset.get("source_url") or ""
            suffix = f" · {note}" if note else ""
            lines.append(
                f"- {asset.get('original_name') or asset.get('name')}"
                f" ({int(asset.get('size') or 0)} bytes){suffix}"
            )
    lines.append("")
    return "\n".join(lines)


def _write_project(project: dict) -> None:
    root = _project_dir(project["id"])
    root.mkdir(parents=True, exist_ok=True)
    (root / "assets").mkdir(exist_ok=True)
    (root / "outputs").mkdir(exist_ok=True)
    project["user_inputs"] = user_inputs_snapshot(project)
    (root / "project.json").write_text(
        json.dumps(project, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    (root / "TASK.md").write_text(_task_markdown(project), encoding="utf-8")
    (root / "INPUTS.md").write_text(_inputs_markdown(project), encoding="utf-8")


def create_project(
    module: str,
    title: str,
    brief: str,
    track: str = "",
    animations: str = "",
    character_type: str = "",
    anatomy_confirmed: bool = False,
    scene_scope: str = "",
    core_interaction: str = "",
    owner_id: str = "",
    schedule_hours: int = 6,
    game_slug: str = "",
) -> dict:
    if module not in MODULES:
        raise StudioError("未知创作模块")
    import curriculum
    import users as user_store

    owner = user_store.get_user(owner_id) if owner_id else None
    allowed, reason = curriculum.can_use_module(owner, module, (track or "").strip())
    if not allowed:
        raise StudioError(reason)
    brief = (brief or "").strip()
    if not brief:
        raise StudioError("请描述你想实现的想法")
    project_id = _new_id()
    while (PROJECTS_DIR / project_id).exists():
        project_id = _new_id()
    now = time.time()
    project = {
        "id": project_id,
        "module": module,
        "title": (title or "").strip()[:120] or MODULES[module]["title"],
        "brief": brief[:8000],
        "track": (track or "").strip()[:40],
        "animations": (animations or "").strip()[:500],
        "character_type": (character_type or "").strip().lower()[:40],
        "anatomy_confirmed": bool(anatomy_confirmed),
        "scene_scope": (scene_scope or "").strip()[:500],
        "core_interaction": (core_interaction or "").strip()[:500],
        "schedule_hours": _schedule_hours(schedule_hours) if module == "idea-watch" else 0,
        "status": "collecting",
        "created": now,
        "updated": now,
        "assets": [],
        "outputs": [],
        "owner_id": (owner_id or "").strip(),
        "game_slug": (game_slug or "").strip()[:64],
        "module_info": MODULES[module],
        "validation": {},
    }
    project["validation"] = validate_project(project)
    _write_project(project)
    return project


def get_project(project_id: str) -> Optional[dict]:
    try:
        path = _project_dir(project_id) / "project.json"
    except StudioError:
        return None
    if not path.exists():
        return None
    project = json.loads(path.read_text(encoding="utf-8"))
    project["module_info"] = MODULES.get(project["module"], {})
    project["validation"] = validate_project(project)
    if not project.get("user_inputs"):
        project["user_inputs"] = user_inputs_snapshot(project)
        try:
            _write_project(project)
        except OSError:
            pass
    return project


def clone_project(
    source_id: str,
    owner_id: str,
    *,
    copy_assets: bool = True,
    title: str = "",
) -> dict:
    """One-click remake: copy prompt + settings (+ assets) into a new project."""
    source = get_project(source_id)
    if source is None:
        raise StudioError("源项目不存在")
    owner = (owner_id or "").strip()
    if not owner:
        raise StudioError("请先登录后再复刻")
    source_owner = (source.get("owner_id") or "").strip()
    if source_owner and source_owner != owner:
        raise StudioError("只能复刻自己的任务记录")
    base_title = (title or "").strip() or (source.get("title") or MODULES[source["module"]]["title"])
    if "复刻" not in base_title:
        base_title = f"{base_title}（复刻）"
    project = create_project(
        source["module"],
        base_title[:120],
        source.get("brief") or "",
        track=source.get("track") or "",
        animations=source.get("animations") or "",
        character_type=source.get("character_type") or "",
        anatomy_confirmed=bool(source.get("anatomy_confirmed")),
        scene_scope=source.get("scene_scope") or "",
        core_interaction=source.get("core_interaction") or "",
        game_slug=source.get("game_slug") or "",
        owner_id=owner,
        schedule_hours=source.get("schedule_hours") or 6,
    )
    project["cloned_from"] = source_id
    project["cloned_at"] = time.time()
    if copy_assets and source.get("assets"):
        src_assets = _project_dir(source_id) / "assets"
        dst_assets = _project_dir(project["id"]) / "assets"
        dst_assets.mkdir(parents=True, exist_ok=True)
        copied = []
        total = 0
        for asset in source["assets"]:
            name = asset.get("name") or ""
            src = src_assets / name
            if not name or not src.is_file():
                continue
            data = src.read_bytes()
            if total + len(data) > MAX_PROJECT_BYTES:
                break
            dest_name = name
            n = 2
            while (dst_assets / dest_name).exists():
                stem = Path(name).stem
                dest_name = f"{stem}-{n}{Path(name).suffix}"
                n += 1
            shutil.copy2(src, dst_assets / dest_name)
            row = dict(asset)
            row["name"] = dest_name
            row["size"] = len(data)
            copied.append(row)
            total += len(data)
        project["assets"] = copied
    project["validation"] = validate_project(project)
    project["status"] = "ready" if project["validation"]["ready"] else "collecting"
    project["updated"] = time.time()
    _write_project(project)
    return get_project(project["id"]) or project


def list_projects(limit: int = 30, owner_id: Optional[str] = None) -> list[dict]:
    projects = []
    for path in PROJECTS_DIR.glob("*/project.json"):
        try:
            project = json.loads(path.read_text(encoding="utf-8"))
            project["module_info"] = MODULES.get(project["module"], {})
            project["validation"] = validate_project(project)
            if not project.get("user_inputs"):
                project["user_inputs"] = user_inputs_snapshot(project)
        except (OSError, json.JSONDecodeError, KeyError):
            continue
        if owner_id and project.get("owner_id") != owner_id:
            continue
        projects.append(project)
    projects.sort(key=lambda p: p.get("updated", 0), reverse=True)
    return projects[:limit]


def update_project_config(
    project_id: str,
    *,
    title: Optional[str] = None,
    brief: Optional[str] = None,
    track: Optional[str] = None,
    animations: Optional[str] = None,
    character_type: Optional[str] = None,
    anatomy_confirmed: Optional[bool] = None,
    scene_scope: Optional[str] = None,
    core_interaction: Optional[str] = None,
) -> dict:
    project = get_project(project_id)
    if project is None:
        raise StudioError("项目不存在")
    if project.get("status") in {"queued", "running"}:
        raise StudioError("项目正在构建，暂时不能修改配置")
    if title is not None:
        project["title"] = title.strip()[:120] or MODULES[project["module"]]["title"]
    if brief is not None:
        value = brief.strip()
        if not value:
            raise StudioError("创作描述不能为空")
        project["brief"] = value[:8000]
    if track is not None:
        value = track.strip()
        if project["module"] == "style-image" and value not in {"spdesign", "dtdesign"}:
            raise StudioError("请选择 spDesign 或 DTdesign")
        project["track"] = value[:40]
    if animations is not None:

def add_assets(project_id: str, files: Iterable[tuple[str, bytes, str]]) -> dict:
    project = get_project(project_id)
    if project is None:
        raise StudioError("项目不存在")
    root = _project_dir(project_id)
    assets_dir = root / "assets"
    assets_dir.mkdir(exist_ok=True)
    total = sum(int(a.get("size", 0)) for a in project["assets"])
    for original_name, data, content_type in files:
        if not data:
            continue
        suffix = Path(original_name or "").suffix.lower()
        allowed = ALLOWED_EXTENSIONS.get(project["module"], set())
        if suffix not in allowed:
            readable = "、".join(sorted(allowed))
            raise StudioError(f"{project['module_info']['title']} 不接受 {suffix or '无扩展名'} 文件；允许：{readable}")
        if len(data) > MAX_ASSET_BYTES:
            raise StudioError("单个素材不能超过 20MB")
        if total + len(data) > MAX_PROJECT_BYTES:
            raise StudioError("项目素材总量不能超过 80MB")
        safe = _safe_name(original_name)
        candidate = safe
        n = 2
        while (assets_dir / candidate).exists():
            p = Path(safe)
            candidate = f"{p.stem}-{n}{p.suffix}"
            n += 1
        (assets_dir / candidate).write_bytes(data)
        project["assets"].append(
            {
                "name": candidate,
                "original_name": original_name,
                "content_type": content_type or "application/octet-stream",
                "size": len(data),
            }
        )
        total += len(data)
    project["updated"] = time.time()
    if project.get("module") == "game":
        project.pop("target_approval", None)
    project["status"] = "ready" if validate_project(project)["ready"] else "collecting"
    project["validation"] = validate_project(project)
    _write_project(project)
    return project


class _NoRedirect(urllib.request.HTTPRedirectHandler):
    def redirect_request(self, req, fp, code, msg, headers, newurl):
        return None


def _validate_public_reference_url(raw_url: str) -> str:
    value = (raw_url or "").strip()
    if not value:
        raise StudioError("参考网址不能为空")
    if "://" not in value:
        value = "https://" + value
    parsed = urllib.parse.urlsplit(value)
    if parsed.scheme not in {"http", "https"} or not parsed.hostname:
        raise StudioError("参考网址只支持 http 或 https")
    if parsed.username or parsed.password:
        raise StudioError("参考网址不能包含账号密码")
    host = parsed.hostname.rstrip(".").lower()
    if host in {"localhost", "localhost.localdomain"} or host.endswith(".local"):
        raise StudioError("参考网址不能访问本机或内网")
    try:
        addresses = {
            item[4][0]
            for item in socket.getaddrinfo(host, parsed.port or (443 if parsed.scheme == "https" else 80))
        }
    except socket.gaierror as exc:
        raise StudioError(f"参考网址无法解析：{host}") from exc
    if not addresses:
        raise StudioError(f"参考网址无法解析：{host}")
    for address in addresses:
        ip = ipaddress.ip_address(address)
        if not ip.is_global:
            raise StudioError("参考网址不能访问本机、内网或保留地址")
    return urllib.parse.urlunsplit(parsed)


def _reference_filename(url: str, content_type: str, index: int) -> str:
    path_name = Path(urllib.parse.unquote(urllib.parse.urlsplit(url).path)).name
    suffix = Path(path_name).suffix.lower()
    mime = (content_type or "").split(";", 1)[0].strip().lower()
    mime_suffix = {
        "text/html": ".html",
        "text/plain": ".txt",
        "text/markdown": ".md",
        "image/png": ".png",
        "image/jpeg": ".jpg",
        "image/webp": ".webp",
    }.get(mime)
    suffix = mime_suffix or suffix
    if suffix not in REFERENCE_EXTENSIONS | IMAGE_EXTENSIONS:
        guessed = mimetypes.guess_extension(mime) or ""
        suffix = guessed if guessed in REFERENCE_EXTENSIONS | IMAGE_EXTENSIONS else ".html"
    stem = _safe_name(Path(path_name).stem or f"reference-{index}")
    return f"url-{index}-{stem}{suffix}"


def _download_reference_url(raw_url: str, index: int) -> tuple[str, bytes, str, str]:
    current = _validate_public_reference_url(raw_url)
    opener = urllib.request.build_opener(_NoRedirect)
    for _ in range(4):
        request = urllib.request.Request(
            current,
            headers={
                "User-Agent": "IdeaLab-ReferenceFetcher/1.0",
                "Accept": "text/html,text/plain,image/png,image/jpeg,image/webp;q=0.9",
            },
        )
        try:
            response = opener.open(request, timeout=12)
        except urllib.error.HTTPError as exc:
            if exc.code in {301, 302, 303, 307, 308}:
                location = exc.headers.get("Location", "")
                if not location:
                    raise StudioError("参考网址跳转缺少目标地址") from exc
                current = _validate_public_reference_url(
                    urllib.parse.urljoin(current, location)
                )
                continue
            raise StudioError(f"参考网址返回 HTTP {exc.code}") from exc
        except (urllib.error.URLError, TimeoutError, OSError) as exc:
            raise StudioError(f"参考网址读取失败：{exc}") from exc
        with response:
            content_type = response.headers.get("Content-Type", "").lower()
            mime = content_type.split(";", 1)[0].strip()
            if mime not in {
                "text/html",
                "text/plain",
                "text/markdown",
                "image/png",
                "image/jpeg",
                "image/webp",
            }:
                raise StudioError(f"参考网址内容类型不支持：{mime or '未知'}")
            declared = int(response.headers.get("Content-Length") or 0)
            if declared > MAX_REFERENCE_BYTES:
                raise StudioError("单个网址参考不能超过 8MB")
            data = response.read(MAX_REFERENCE_BYTES + 1)
            if len(data) > MAX_REFERENCE_BYTES:
                raise StudioError("单个网址参考不能超过 8MB")
            final_url = _validate_public_reference_url(response.geturl())
            return (
                _reference_filename(final_url, content_type, index),
                data,
                content_type,
                final_url,
            )
    raise StudioError("参考网址跳转次数过多")


def add_reference_urls(project_id: str, urls: Iterable[str]) -> dict:
    project = get_project(project_id)
    if project is None:
        raise StudioError("项目不存在")
    values = list(dict.fromkeys((url or "").strip() for url in urls if (url or "").strip()))
    if not values:
        raise StudioError("请填写至少一个参考网址")
    if len(values) > MAX_REFERENCE_URLS:
        raise StudioError(f"每次最多添加 {MAX_REFERENCE_URLS} 个参考网址")
    downloaded = [_download_reference_url(url, index) for index, url in enumerate(values, 1)]
    project = add_assets(
        project_id,
        ((name, data, content_type) for name, data, content_type, _ in downloaded),
    )
    start = len(project["assets"]) - len(downloaded)
    for asset, (_, _, _, source_url) in zip(project["assets"][start:], downloaded):
        asset["source_url"] = source_url
        asset["source_type"] = "url"
    project["updated"] = time.time()
    project["validation"] = validate_project(project)
    _write_project(project)
    return project

def add_output(project_id: str, name: str, page_id: str = "") -> dict:
    project = get_project(project_id)
    if project is None:
        raise StudioError("项目不存在")
    now = time.time()
    reused = False
    if page_id:
        for item in project["outputs"]:
            if item.get("page_id") == page_id:
                item["name"] = name[:120]
                item["created"] = now
                reused = True
                break
    if not reused:
        project["outputs"].append(

        )
    project["status"] = "completed"
    project["updated"] = time.time()
    _write_project(project)
    _archive_and_unlock(project)
    return project

        return

    import library

    outputs = _project_dir(project["id"]) / "outputs"
    page_id = ""
    for item in reversed(project.get("outputs") or []):
        if item.get("page_id"):
            page_id = item["page_id"]
            break
    added = library.harvest_outputs(
        owner_id,
        project["id"],
        project.get("module") or "",

        outputs,
        page_id=page_id,
    )

def continue_prompt(project: dict, previous: str = "") -> str:
    snippet = (previous or "").strip().replace("\n", " ")[:240]
    if project.get("module") == "adventure-reader":
        return (
            "上一轮被截断。请用工具写完 outputs/book.json，再跑校验和填空自测。\n"
            "不要把整份 JSON 贴进对话。材料在 assets/source.txt。\n"
            f"需求摘要：{(project.get('brief') or '')[:400]}\n"
            + (f"上次截断前的摘要：{snippet}\n" if snippet else "")
        )
    return (
        "上一轮因 max-tokens 被截断，对话里的半成品不算完成。\n"
        "请立刻用工具写入或补全 outputs/preview.html："
        "必须是可在沙箱 iframe 运行的自包含单文件，无构建步骤、无服务端。\n"
        "不要重写长计划，不要把完整源码贴进回复。写完后只报告该文件的绝对路径。\n"
        f"模块：{project.get('module') or 'text'}。需求：{(project.get('brief') or '')[:400]}\n"
        + (f"上次截断前的摘要：{snippet}\n" if snippet else "")
    )


def update_status(project_id: str, status: str, run_id: str = "", error: str = "") -> dict:
    project = get_project(project_id)
    if project is None:
        raise StudioError("项目不存在")
    project["status"] = status
    project["updated"] = time.time()
    if run_id:
        project["run_id"] = run_id
    if error:
        project["error"] = error[:1000]
    elif "error" in project:
        project.pop("error")
    _write_project(project)
    if status == "completed":
        _archive_and_unlock(project)
    return project


def _task_markdown(project: dict) -> str:
    info = MODULES[project["module"]]
    validation = validate_project(project)
    asset_lines = "\n".join(
        f"- `assets/{a['name']}` ({a['content_type']}, {a['size']} bytes)"
        + (f"\n  - 来源：{a['source_url']}" if a.get("source_url") else "")
        for a in project.get("assets", [])
    ) or "- 尚未上传"
    gate_lines = "\n".join(f"- {g}" for g in validation["gates"]) or "- 无"
    issue_lines = "\n".join(f"- {x}" for x in validation["issues"]) or "- 已满足"
    return f"""# {project['title']}

## 模块

{info['title']} · `{info['skill']}`

## 想法

{project['brief']}

## 输入素材

{asset_lines}

## 执行门槛

{gate_lines}

## 当前缺项

{issue_lines}

## Harness Agent 指令

```text
{agent_prompt(project)}
```
"""
