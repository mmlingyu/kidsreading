def default_set() -> set[str]:
    return set(DEFAULT_SKILLS)


def unlocked_set(user) -> set[str]:
    names = set(DEFAULT_SKILLS)
    if user is not None:
        names.update(user.unlocked_skills or [])
    return names


def is_unlocked(user, skill: str) -> bool:
    meta = SKILL_META.get(skill)
    if meta is None:
        return True
    if meta["availability"] == "default":
        return True
    return skill in unlocked_set(user)


def required_skills(module: str, track: str = "") -> list[str]:
    if module == "style-image" and track in {"spdesign", "dtdesign"}:
        return [track]
    return list(MODULE_SKILLS.get(module, []))


def can_use_module(user, module: str, track: str = "") -> tuple[bool, str]:
    if user is None:
        return False, "请先人脸登录后再开始创作。生成完成后作品会进入你的资产库。"
    needed = required_skills(module, track)
    if not needed:
        return True, ""
    if all(is_unlocked(user, name) for name in needed):
        return True, ""
    if module == "style-image" and (
        is_unlocked(user, "spdesign") or is_unlocked(user, "dtdesign")
    ):
        if track and not is_unlocked(user, track):
            title = SKILL_META.get(track, {}).get("title", track)
            return False, f"先答题或完成任务解锁 {title}"
        return True, ""
    locked = [SKILL_META.get(name, {}).get("title") or name for name in needed if not is_unlocked(user, name)]
    task = SKILL_META.get(needed[0], {}).get("task") or "完成对应任务或答题"
    return False, f"尚未解锁：{' / '.join(locked)}。{task}，或打开答题。"


def quiz_public(skill: str) -> Optional[dict]:
    quiz = QUIZZES.get(skill)
    if quiz is None:
        return None
    return {
        "skill": quiz["skill"],
        "title": quiz["title"],
        "pass": quiz["pass"],
        "total": len(quiz["questions"]),
        "questions": [
            {"id": q["id"], "prompt": q["prompt"], "options": q["options"]}
            for q in quiz["questions"]
        ],
    }


def grade_quiz(skill: str, answers: dict) -> dict:
    quiz = QUIZZES.get(skill)
    if quiz is None:
        raise ValueError("没有这门解锁测验")
    score = 0
    wrong = []
    for question in quiz["questions"]:
        given = answers.get(question["id"])
        try:
            index = int(given)
        except (TypeError, ValueError):
            index = -1
        if index == question["answer"]:
            score += 1
        else:
            wrong.append(question["id"])
    return {
        "skill": skill,
        "score": score,
        "total": len(quiz["questions"]),
        "pass_score": quiz["pass"],
        "passed": score >= quiz["pass"],
        "wrong": wrong,
    }


def catalog(user) -> dict:
    owned = unlocked_set(user)
    skills = []
    for name, meta in SKILL_META.items():
        skills.append(
            {
                "name": name,
                "title": meta["title"],
                "module": meta["module"],
                "availability": meta["availability"],
                "unlocked": name in owned,
                "task": meta.get("task") or "",
                "has_quiz": name in QUIZZES,
            }
        )
    return {
        "default": list(DEFAULT_SKILLS),
        "unlocked": sorted(owned),
        "skills": skills,
        "logged_in": user is not None,
    }


def refresh_unlocks(user_id: str) -> list[str]:
    import studio
    import users

    user = users.get_user(user_id)
    if user is None:
        return []
    completed = {
        project.get("module")
        for project in studio.list_projects(limit=200, owner_id=user_id)
        if project.get("status") == "completed" and project.get("module")
    }
    gained = []
    for name, meta in SKILL_META.items():
        if meta["availability"] != "unlockable" or name in user.unlocked_skills:
            continue
        need = set(meta.get("complete_any") or [])
        if need and completed.intersection(need):
            gained.append(name)
    if gained:
        users.unlock_skills(user_id, gained)
    return gained
