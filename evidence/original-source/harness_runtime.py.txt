# Catalog ids that declare image input. `deepseek-v4-flash` is text-only.
# `deepseek-flash` is DeepSeek-V4.1-Flash (text + image). The retired
# `deepseek-v4-flash-vision-exp` id still advertises images.
DEFAULT_MODEL = "deepseek-flash"
VISION_MODELS = {"deepseek-flash", "deepseek-v4-flash-vision-exp"}
TEXT_ONLY_LEGACY_DEFAULT = "deepseek-v4-flash"


def resolved_model() -> str:
    raw = os.environ.get("DSH_MODEL", "").strip()
    if not raw or raw == TEXT_ONLY_LEGACY_DEFAULT:
        return DEFAULT_MODEL
    return raw


def model_supports_vision(model: str) -> bool:
    name = (model or "").strip().lower()
    return name in VISION_MODELS or "vision" in name


def _runtime_patches() -> list[str]:
    patches = [
        str(Path(item.strip()).resolve())
        for item in os.environ.get("DSH_PATCHES", "").split(os.pathsep)
        if item.strip()
    ]
    command = os.environ.get("BLENDER_MCP_COMMAND", "").strip()
    if command:
        try:
            args = json.loads(os.environ.get("BLENDER_MCP_ARGS", "[]"))
        except json.JSONDecodeError as exc:
            raise RuntimeError("BLENDER_MCP_ARGS 必须是 JSON 字符串数组") from exc
        if not isinstance(args, list) or not all(isinstance(item, str) for item in args):
            raise RuntimeError("BLENDER_MCP_ARGS 必须是 JSON 字符串数组")
        patch = skill_registry.DSH_HOME / "studio-blender-mcp.patch.yml"
        patch.parent.mkdir(parents=True, exist_ok=True)
        patch.write_text(
            "- insert:\n"
            "    - id: studio-blender-mcp\n"
            "      name: '@deepseek-ai/dsh-mcp-client'\n"
            "      config:\n"
            "        serverName: blender\n"
            "        transport: stdio\n"
            f"        command: {json.dumps(command)}\n"
            f"        args: {json.dumps(args, ensure_ascii=False)}\n",
            encoding="utf-8",
        )
        patches.append(str(patch.resolve()))
    if os.environ.get("REA_MCP", "1").strip().lower() not in {"0", "false", "off", "no"}:
        raw_args = os.environ.get("REA_MCP_ARGS", "").strip()
        if raw_args:
            try:
                rea_args = json.loads(raw_args)
            except json.JSONDecodeError as exc:
                raise RuntimeError("REA_MCP_ARGS 必须是 JSON 字符串数组") from exc
            if not isinstance(rea_args, list) or not all(isinstance(item, str) for item in rea_args):
                raise RuntimeError("REA_MCP_ARGS 必须是 JSON 字符串数组")
        else:
            rea_args = ["-y", "rea-agents@6.8.0", "mcp"]
        command = os.environ.get("REA_MCP_COMMAND", "npx").strip() or "npx"
        patch = skill_registry.DSH_HOME / "studio-rea-mcp.patch.yml"
        patch.parent.mkdir(parents=True, exist_ok=True)
        patch.write_text(
            "- insert:\n"
            "    - id: studio-rea-mcp\n"
            "      name: '@deepseek-ai/dsh-mcp-client'\n"
            "      config:\n"
            "        serverName: rea\n"
            "        transport: stdio\n"
            f"        command: {json.dumps(command)}\n"
            f"        args: {json.dumps(rea_args, ensure_ascii=False)}\n",
            encoding="utf-8",
        )
        patches.append(str(patch.resolve()))
    return patches


def capabilities() -> dict:
    sdk = importlib.util.find_spec("deepseek_harness") is not None
    key = bool(os.environ.get("DEEPSEEK_API_KEY", "").strip())
    dashscope_key = bool(os.environ.get("DASHSCOPE_API_KEY", "").strip())
    dashscope_sdk = importlib.util.find_spec("dashscope") is not None
    blender = bool(os.environ.get("BLENDER_MCP_COMMAND", "").strip())
    try:
        patches = _runtime_patches()
        patch_error = ""
    except RuntimeError as exc:
        patches = []
        patch_error = str(exc)
    execution_mode = os.environ.get("HARNESS_EXECUTION_MODE", "disabled").strip()
    user_patches = [
        item
        for item in patches
        if not item.endswith("studio-rea-mcp.patch.yml")
    ]
    safe_execution = execution_mode == "trusted-local" or (
        execution_mode == "isolated" and bool(user_patches)
    )
    return {
        "harness_sdk": sdk,
        "deepseek_credentials": key,
        "agent_runtime": sdk and key,
        "provider": os.environ.get("DSH_PROVIDER", "deepseek-official"),
        "model": resolved_model(),
        "vision": model_supports_vision(resolved_model()),
        "dashscope_image": dashscope_key and dashscope_sdk,
        "blender_gateway": blender,
        "knowledge_poetry": True,
        "knowledge_tutors": True,
        "skills": len(skill_registry.list_skills()),
        "patches": patches,
        "configuration_error": patch_error,
        "execution_mode": execution_mode,
        "safe_execution": safe_execution,
    }


def explain_finish_reason(reason: Optional[str]) -> str:
    if not reason:
        return "Agent 没有返回完成原因。"
    return FINISH_REASON_HELP.get(reason, f"结束原因：{reason}")


def extract_runtime_error(events: Optional[list]) -> str:
    """Pull the concrete runtime/model error out of turn/end (or similar) events."""
    if not events:
        return ""
    for event in reversed(events):
        if not isinstance(event, dict):
            continue
        candidates: list[object] = [event]
        payload = event.get("payload")
        if isinstance(payload, dict):
            candidates.append(payload)
            nested = payload.get("event")
            if isinstance(nested, dict):
                candidates.append(nested)
        for item in candidates:
            if not isinstance(item, dict):
                continue
            data = item.get("data") if isinstance(item.get("data"), dict) else item
            if not isinstance(data, dict):
                continue
            reason = data.get("reason")
            if not isinstance(reason, dict):
                continue
            err = reason.get("error")
            if isinstance(err, dict):
                message = str(err.get("message") or "").strip()
                if message:
                    return message.replace("\r\n", "\n")
            if reason.get("kind") == "error":
                message = str(reason.get("message") or "").strip()
                if message:
                    return message.replace("\r\n", "\n")
    return ""


def should_continue_truncated(
    finish_reason: Optional[str],
    has_artifact: bool,
    used: int,
    limit: int = MAX_TOKEN_CONTINUES,
) -> bool:
    return (finish_reason or "") == "max-tokens" and not has_artifact and 0 <= used < limit


def settle_agent_result(
    finish_reason: Optional[str],
    *,
    cancelled: bool,
    has_artifact: bool,
    runtime_error: str = "",
) -> dict:
    if cancelled:
        return {
            "status": "cancelled",
            "label": "任务已取消",
            "detail": "可以重新开始构建",
            "error": "",
        }
    if has_artifact:
        truncated = (finish_reason or "") == "max-tokens"
        return {
            "status": "completed",
            "label": "作品已完成" if not truncated else "已导入现有产物",
            "detail": (
                "可以直接打开安全预览"
                if not truncated
                else "输出曾被截断，已导入当前成品"
            ),
            "error": "",
        }
    if finish_reason == "completed":
        return {
            "status": "failed",
            "label": "未生成可交付产物",
            "detail": "Agent 已结束，但缺少该 Skill 契约要求的最终产物",
            "error": "Agent 已结束，但没有生成可通过 Review 的最终产物",
        }
    message = explain_finish_reason(finish_reason)
    detail = (runtime_error or "").strip()
    if detail:
        # Keep the friendly summary, but surface the concrete runtime fault.
        if "koffi.node" in detail.lower() or "specified module could not be found" in detail.lower():
            message = (
                "Harness 运行时原生模块加载失败（koffi）。"
                "请重新运行 scripts/ensure_harness_native.ps1 或重启 IdeaLab。"
            )
        return {
            "status": "failed",
            "label": "Agent 未完成构建",
            "detail": message,
            "error": f"{message}\n详情：{detail[:1500]}",
        }
    return {
        "status": "failed",
        "label": "Agent 未完成构建",
        "detail": message,
        "error": message,
    }


class HarnessRunManager:
    """Owns background SDK runs and a persisted event projection."""

    def __init__(self) -> None:
        self._runs: dict[str, dict] = {}
        self._tasks: dict[str, asyncio.Task] = {}
        self._harnesses: dict[str, object] = {}
        self._lock = threading.RLock()
        self._load_runs()

    def _load_runs(self) -> None:
        for path in RUNS_DIR.glob("*.json"):
            try:
                run = json.loads(path.read_text(encoding="utf-8"))
                if run.get("status") in {"queued", "running", "cancel_requested"}:
                    run["status"] = "interrupted"
                    run["error"] = "服务重启中断了任务"
                    run["progress"] = {
                        "phase": "failed",
                        "percent": run.get("progress", {}).get("percent", 0),
                        "label": "任务被服务重启中断",
                        "detail": "可以重新开始构建",
                    }
                    try:
                        studio.update_status(
                            run["project_id"], "interrupted", run["id"], run["error"]
                        )
                    except (KeyError, studio.StudioError):
                        pass
                run.setdefault(
                    "progress",
                    {
                        "phase": "completed" if run.get("status") == "completed" else "queued",
                        "percent": 100 if run.get("status") == "completed" else 0,
                        "label": "构建已完成" if run.get("status") == "completed" else "等待开始",
                        "detail": "",
                    },
                )
                run.setdefault("started_at", run.get("created"))
                if run.get("status") in {
                    "completed",
                    "failed",
                    "blocked",
                    "cancelled",
                    "interrupted",
                    "review_failed",
                    "awaiting_approval",
                }:
                    run.setdefault("finished_at", run.get("updated"))
                    run.setdefault(
                        "duration_seconds",
                        max(
                            0,
                            float(run.get("updated", 0))
                            - float(run.get("created", 0)),
                        ),
                    )
                run.setdefault("phase_durations", {})
                run.setdefault(
                    "phase_started_at",
                    run.get("created"),
                )
                existing_usage = run.get("usage")
                if not isinstance(existing_usage, dict) or not existing_usage.get(
                    "steps"
                ):
                    run["usage"] = run_stats.usage_from_events(run.get("events"))
                else:
                    run["usage"] = existing_usage
                self._runs[run["id"]] = run
                self._persist(run)
            except (OSError, json.JSONDecodeError, KeyError):
                continue
        latest_by_project: dict[str, dict] = {}
        for run in self._runs.values():
            project_id = run.get("project_id", "")
            current = latest_by_project.get(project_id)
            if project_id and (
                current is None or run.get("created", 0) > current.get("created", 0)
            ):
                latest_by_project[project_id] = run
        for project_id, run in latest_by_project.items():
            try:
                studio.update_status(
                    project_id,
                    run.get("status", "interrupted"),
                    run["id"],
                    run.get("error", ""),
                )
            except studio.StudioError:
                pass

    def _persist(self, run: dict) -> None:
        (RUNS_DIR / f"{run['id']}.json").write_text(
            json.dumps(run, ensure_ascii=False, indent=2), encoding="utf-8"
        )

    def get(self, run_id: str) -> Optional[dict]:
        with self._lock:
            run = self._runs.get(run_id)
            return json.loads(json.dumps(run)) if run else None

    def list(self, project_id: str = "") -> list[dict]:
        with self._lock:
            runs = list(self._runs.values())
        if project_id:
            runs = [r for r in runs if r.get("project_id") == project_id]
        runs.sort(key=lambda r: r.get("created", 0), reverse=True)
        return json.loads(json.dumps(runs))

    def events_since(self, run_id: str, cursor: int) -> tuple[list[dict], int, str]:
        with self._lock:
            run = self._runs.get(run_id)
            if run is None:
                return [], cursor, "missing"
            events = run.get("events", [])
            return json.loads(json.dumps(events[cursor:])), len(events), run["status"]

    async def start(self, project_id: str) -> dict:
        project = studio.get_project(project_id)
        if project is None:
            raise studio.StudioError("项目不存在")
        validation = studio.validate_project(project)
        if not validation["ready"]:
            raise studio.StudioError("；".join(validation["issues"]))
        project = studio.prepare_project_inputs(project_id)

        skill_registry.sync_skills()
        run_id = uuid.uuid4().hex[:14]
        now = time.time()
        inputs = project.get("user_inputs") or studio.user_inputs_snapshot(project)
        run = {
            "id": run_id,
            "project_id": project_id,
            # Unique per run: the runtime persists sessions in the shared
            # DSH_HOME, so a project-only id collides on a project's 2nd run
            # ("session ... already exists"). run_id is already unique.
            "session_id": f"studio-{project_id}-{run_id}",
            "status": "queued",
            "created": now,
            "updated": now,
            "module": project.get("module", ""),
            "track": project.get("track", ""),
            "user_inputs": inputs,
            "prompt": inputs.get("prompt") or inputs.get("brief") or "",
            "started_at": None,
            "finished_at": None,
            "duration_seconds": None,
            "phase_started_at": now,
            "phase_durations": {},
            "events": [],
            "usage": run_stats.empty_usage(),
            "final_response": "",
            "finish_reason": None,
            "page_id": "",
            "error": "",
            "progress": {
                "phase": "queued",
                "percent": 5,
                "label": "任务已进入队列",
                "detail": "正在准备 Agent 工作区",
            },
            "capabilities": capabilities(),
        }
        with self._lock:
            self._runs[run_id] = run
            self._persist(run)
        self._append(run_id, {"kind": "progress", **run["progress"]})
        studio.update_status(project_id, "queued", run_id)

        missing = []
        if not run["capabilities"]["harness_sdk"]:
            missing.append("deepseek-harness-sdk")
        if not run["capabilities"]["deepseek_credentials"]:
            missing.append("DEEPSEEK_API_KEY")
        if not run["capabilities"]["safe_execution"]:
            missing.append(
                "HARNESS_EXECUTION_MODE=trusted-local，或 isolated + DSH_PATCHES"
            )
        if run["capabilities"]["configuration_error"]:
            missing.append(run["capabilities"]["configuration_error"])
        if project["module"] == "style-image" and not run["capabilities"]["dashscope_image"]:
            missing.append("DashScope 图像工具")
        if project["module"] in {"style-image", "game", "animation"} and not run[
            "capabilities"
        ].get("vision"):
            missing.append(
                f"支持看图的 DeepSeek 模型（当前 {run['capabilities']['model']}，"
                "请设 DSH_MODEL=deepseek-flash）"
            )
        if project["module"] == "game" and not run["capabilities"]["blender_gateway"]:
            missing.append("Blender MCP 网关")
        if missing:
            self._fail(run_id, "Agent Runtime 不可用：" + "、".join(missing))
            return self.get(run_id) or run

        task = asyncio.create_task(asyncio.to_thread(self._execute, run_id))
        self._tasks[run_id] = task
        task.add_done_callback(lambda _task: self._tasks.pop(run_id, None))
        return self.get(run_id) or run

    def _append(self, run_id: str, event: dict) -> None:
        with self._lock:
            run = self._runs[run_id]
            event["seq"] = len(run["events"])
            event["time"] = time.time()
            run["events"].append(event)
            run["updated"] = time.time()
            self._persist(run)

    def _accumulate_usage(
        self, run_id: str, payload: dict | None, event_type: str
    ) -> None:
        delta = run_stats.extract_assistant_usage(payload, event_type)
        if not delta:
            return
        with self._lock:
            run = self._runs[run_id]
            usage = run.get("usage")
            if not isinstance(usage, dict):
                usage = run_stats.empty_usage()
            run_stats.apply_usage_delta(usage, delta)
            run["usage"] = usage
            steps = int(usage.get("steps") or 0)
            emit = steps == 1 or steps % 10 == 0
            if emit:
                label = run_stats.format_usage_label(usage)
                event = {
                    "kind": "usage",
                    "text": label,
                    "usage_label": label,
                    "usage": {
                        key: usage.get(key, 0) for key in run_stats.USAGE_KEYS
                    }
                    | {"steps": steps},
                    "seq": len(run["events"]),
                    "time": time.time(),
                }
                run["events"].append(event)
            run["updated"] = time.time()
            self._persist(run)

    def _emit_usage_summary(self, run_id: str) -> None:
        with self._lock:
            run = self._runs.get(run_id)
            if run is None:
                return
            usage = run.get("usage")
            if not isinstance(usage, dict):
                usage = run_stats.usage_from_events(run.get("events"))
                run["usage"] = usage
            if not int(usage.get("steps") or 0) and not any(
                int(usage.get(key) or 0) for key in run_stats.USAGE_KEYS
            ):
                return
            label = run_stats.format_usage_label(usage)
            event = {
                "kind": "usage",
                "text": "本次累计 " + label,
                "usage_label": label,
                "usage": {
                    key: usage.get(key, 0) for key in run_stats.USAGE_KEYS
                }
                | {"steps": int(usage.get("steps") or 0)},
                "seq": len(run["events"]),
                "time": time.time(),
            }
            run["events"].append(event)
            run["updated"] = time.time()
            self._persist(run)

    def _set_status(self, run_id: str, status: str) -> None:
        with self._lock:
            run = self._runs[run_id]
            run["status"] = status
            now = time.time()
            run["updated"] = now
            if status == "running" and not run.get("started_at"):
                run["started_at"] = now
            self._persist(run)

    def _set_progress(
        self,
        run_id: str,
        phase: str,
        percent: int,
        label: str,
        detail: str = "",
    ) -> None:
        progress = {
            "phase": phase,
            "percent": max(0, min(100, int(percent))),
            "label": label,
            "detail": detail,
        }
        with self._lock:
            run = self._runs[run_id]
            now = time.time()
            previous = (run.get("progress") or {}).get("phase")
            if previous and previous != phase:
                started = float(run.get("phase_started_at") or run.get("created") or now)
                durations = run.setdefault("phase_durations", {})
                durations[previous] = round(
                    float(durations.get(previous, 0)) + max(0, now - started),
                    3,
                )
                run["phase_started_at"] = now
            run["progress"] = progress
            run["updated"] = now
            if phase in {
                "completed",
                "failed",
                "blocked",
                "cancelled",
                "review_failed",
                "awaiting_approval",
            }:
                run["finished_at"] = now
                run["duration_seconds"] = round(
                    max(0, now - float(run.get("created") or now)),
                    3,
                )
            self._persist(run)
        self._append(run_id, {"kind": "progress", **progress})

    def _fail(self, run_id: str, message: str) -> None:
        current = self.get(run_id) or {}
        percent = current.get("progress", {}).get("percent", 0)
        self._set_progress(run_id, "failed", percent, "构建未能继续", message)
        with self._lock:
            run = self._runs[run_id]
            run["status"] = "blocked"
            run["error"] = message
            run["updated"] = time.time()
            self._persist(run)
            project_id = run["project_id"]
        studio.update_status(project_id, "blocked", run_id, message)
        self._append(run_id, {"kind": "error", "text": message})

    def _execute(self, run_id: str) -> None:
        run = self.get(run_id)
        if run is None:
            return
        project = studio.get_project(run["project_id"])
        if project is None:
            self._fail(run_id, "项目不存在")
            return
        self._set_status(run_id, "running")
        self._set_progress(
            run_id, "prepare", 12, "正在准备素材", "检查项目文件与运行能力"
        )
        studio.update_status(project["id"], "running", run_id)
        self._append(
            run_id,
            {
                "kind": "status",
                "text": f"启动 DeepSeek Harness · {run['capabilities']['model']}",
            },
        )
        try:
            from deepseek_harness import DeepSeekHarness

            root = Path(studio.PROJECTS_DIR) / project["id"]
            harness = DeepSeekHarness(
                dsh_home=str(skill_registry.DSH_HOME.resolve()),
                cwd=str(root.resolve()),
                profile=os.environ.get("DSH_PROFILE", "sdk"),
                patches=tuple(_runtime_patches()),
                provider=os.environ.get("DSH_PROVIDER", "deepseek-official"),
                model=resolved_model(),
                reasoning_effort=os.environ.get("DSH_REASONING_EFFORT") or None,
                max_tokens=int(os.environ.get("DSH_MAX_TOKENS", "32768")),
                initialize_timeout_seconds=INITIALIZE_TIMEOUT_SECONDS,
                request_timeout_seconds=REQUEST_TIMEOUT_SECONDS,
                env={
                    "HTML_SHARE_ROOT": str(BASE_DIR.resolve()),
                    **(
                        {"DSH_PERMISSION_MODE": "danger-full-access"}
                        if os.environ.get("HARNESS_EXECUTION_MODE", "").strip()
                        == "trusted-local"
                        else {}
                    ),
                },
            )
            self._harnesses[run_id] = harness
            self._set_progress(
                run_id,
                "agent",
                25,
                "正在启动 Harness Runtime",
                (
                    f"首次可能下载依赖，最长约 {int(INITIALIZE_TIMEOUT_SECONDS)}s · "
                    f"/{project.get('track') or project['module_info']['skill']}"
                ),
            )

            tool_events = 0
            activity_started = False
            def notify(notification) -> None:
                nonlocal tool_events, activity_started
                payload = notification.payload
                event = payload.get("event") if isinstance(payload, dict) else None
                event_type = event.get("type") if isinstance(event, dict) else ""
                if not activity_started:
                    activity_started = True
                    self._set_progress(
                        run_id, "build", 35, "Agent 正在生成作品", "已开始分析需求"
                    )
                if "tool" in event_type.lower():
                    tool_events += 1
                    self._set_progress(
                        run_id,
                        "build",
                        min(78, 40 + tool_events * 3),
                        "正在调用创作工具",
                        event_type,
                    )
                self._append(
                    run_id,
                    {
                        "kind": "notification",
                        "method": notification.method,
                        "event_type": event_type,
                        "payload": payload,
                    },
                )
                self._accumulate_usage(run_id, payload, event_type)

            prompt = studio.agent_prompt(project)
            continues_used = 0
            result = None
            while True:
                result = harness.run(
                    prompt,
                    session_id=run["session_id"],
                    on_notification=notify,
                )
                with self._lock:
                    target = self._runs[run_id]
                    target["final_response"] = result.final_response or ""
                    target["finish_reason"] = result.finish_reason
                    target["updated"] = time.time()
                    self._persist(target)
                runtime_error = extract_runtime_error(
                    list(getattr(result, "events", None) or [])
                )
                status_text = (
                    f"回合结束 · {result.finish_reason or 'unknown'}"
                    f"：{explain_finish_reason(result.finish_reason)}"
                )
                if runtime_error:
                    status_text = f"{status_text}\n详情：{runtime_error[:500]}"
                self._append(
                    run_id,
                    {
                        "kind": "status",
                        "text": status_text,
                    },
                )
                cancelled = (self.get(run_id) or {}).get("status") == "cancel_requested"
                has_artifact = self._has_importable_output(project)
                if cancelled or not should_continue_truncated(
                    result.finish_reason,
                    has_artifact,
                    continues_used,
                ):
                    break
                continues_used += 1
                logger.warning(
                    "run %s truncated by max-tokens; continuing %s/%s",
                    run_id,
                    continues_used,
                    MAX_TOKEN_CONTINUES,
                )
                resume_target = (
                    "outputs/book.json"
                    if project.get("module") == "adventure-reader"
                    else "outputs/preview.html"
                )
                self._set_progress(
                    run_id,
                    "build",
                    80,
                    "输出被截断，正在续写",
                    f"第 {continues_used}/{MAX_TOKEN_CONTINUES} 次续写。"
                    f"请改用工具写入 {resume_target}，不要把源码贴进对话。",
                )
                self._append(
                    run_id,
                    {
                        "kind": "error",
                        "text": (
                            "max-tokens：回复被截断，尚未产出可预览文件，"
                            f"正在续写（{continues_used}/{MAX_TOKEN_CONTINUES}）。"
                        ),
                    },
                )
                prompt = studio.continue_prompt(project, result.final_response or "")

            game_target = (
                Path(studio.PROJECTS_DIR) / project["id"] / "outputs" / "target.png"
            )
            target_approved = bool(
                (project.get("target_approval") or {}).get("approved")
            )
            if (
                project.get("module") == "game"
                and game_target.is_file()
                and not target_approved
            ):
                self._set_progress(
                    run_id,
                    "awaiting_approval",
                    32,
                    "等待确认目标图",
                    "确认后才会开始 Blender 建模，避免长时间生成错误方向",
                )
                with self._lock:
                    target = self._runs[run_id]
                    target["status"] = "awaiting_approval"
                    target["updated"] = time.time()
                    self._persist(target)
                studio.update_status(project["id"], "awaiting_approval", run_id)
                return

            self._set_progress(
                run_id, "review", 84, "正在检查 Agent 结果", "确认输出与完成状态"
            )
            cancelled = (self.get(run_id) or {}).get("status") == "cancel_requested"
            has_artifact = self._has_importable_output(project)
            review_report = None
            review_attempts = 0
            if has_artifact and not cancelled:
                root = Path(studio.PROJECTS_DIR) / project["id"]
                while True:
                    review_report = quality_review.review_project(project, root)
                    (root / "outputs" / "quality-report.json").write_text(
                        json.dumps(review_report, ensure_ascii=False, indent=2),
                        encoding="utf-8",
                    )
                    with self._lock:
                        target = self._runs[run_id]
                        target["quality_report"] = review_report
                        target["review_attempts"] = review_attempts
                        target["updated"] = time.time()
                        self._persist(target)
                    if review_report["passed"] or review_attempts >= 2:
                        break
                    review_attempts += 1
                    self._set_progress(
                        run_id,
                        "build",
                        86,
                        f"Review 未通过，正在自动修订 {review_attempts}/2",
                        "优先修复一个最大的交付问题",
                    )
                    result = harness.run(
                        quality_review.revision_prompt(
                            project, review_report, review_attempts
                        ),
                        session_id=run["session_id"],
                        on_notification=notify,
                    )
                    with self._lock:
                        target = self._runs[run_id]
                        target["final_response"] = result.final_response or ""
                        target["finish_reason"] = result.finish_reason
                        target["review_attempts"] = review_attempts
                        target["updated"] = time.time()
                        self._persist(target)
                    self._set_progress(
                        run_id,
                        "review",
                        88,
                        "正在复查修订结果",
                        f"第 {review_attempts}/2 次自动修订已结束",
                    )
                    has_artifact = self._has_importable_output(project)
                    if not has_artifact:
                        break
            if review_report is not None and not review_report["passed"]:
                decision = {
                    "status": "review_failed",
                    "label": "产物 Review 未通过",
                    "detail": review_report["errors"][0] if review_report["errors"] else "请查看质量报告",
                    "error": "；".join(review_report["errors"]),
                }
            else:
                decision = settle_agent_result(
                    result.finish_reason if result is not None else None,
                    cancelled=cancelled,
                    has_artifact=has_artifact,
                    runtime_error=extract_runtime_error(
                        list(getattr(result, "events", None) or [])
                        if result is not None
                        else []
                    ),
                )
            page_id = ""
            if decision["status"] == "completed" and has_artifact:
                self._set_progress(
                    run_id, "publish", 92, "正在准备预览", "识别并导入最终产物"
                )
                try:
                    page_id = self._import_preview(project)
                except Exception as exc:
                    tb = traceback.format_exc()
                    logger.exception("import preview failed for run %s", run_id)
                    self._record_failure(
                        run_id,
                        project,
                        f"产物导入失败：{type(exc).__name__}: {exc}",
                        traceback_text=tb,
                    )
                    return
                if page_id:
                    decision["label"] = "作品已完成"
                    decision["detail"] = "可以直接打开安全预览"
            if decision["status"] in {"failed", "review_failed"}:
                snippet = (result.final_response if result else "") or ""
                if snippet:
                    decision["error"] = (
                        f"{decision['error']}\n最后回复摘要：{snippet[:400]}"
                    )
                self._record_failure(
                    run_id,
                    project,
                    decision["error"],
                    finish_reason=result.finish_reason if result else None,
                    status=decision["status"],
                    label=decision["label"],
                    detail=decision["detail"],
                )
                return
            self._set_progress(
                run_id,
                decision["status"],
                100 if decision["status"] == "completed" else (self.get(run_id) or {}).get("progress", {}).get("percent", 0),
                decision["label"],
                decision["detail"],
            )
            with self._lock:
                target = self._runs[run_id]
                target["status"] = decision["status"]
                target["page_id"] = page_id
                target["error"] = decision["error"]
                target["updated"] = time.time()
                self._persist(target)
            studio.update_status(
                project["id"],
                decision["status"],
                run_id,
                decision["error"],
            )
        except Exception as exc:
            current = self.get(run_id) or {}
            if current.get("status") == "cancel_requested":
                percent = current.get("progress", {}).get("percent", 0)
                self._set_progress(
                    run_id, "cancelled", percent, "任务已取消", "可以重新开始构建"
                )
                self._set_status(run_id, "cancelled")
                studio.update_status(project["id"], "cancelled", run_id)
            else:
                tb = traceback.format_exc()
                logger.exception("studio run %s failed", run_id)
                self._record_failure(
                    run_id,
                    project,
                    f"{type(exc).__name__}: {exc}",
                    traceback_text=tb,
                )
        finally:
            self._emit_usage_summary(run_id)
            harness = self._harnesses.pop(run_id, None)
            if harness is not None:
                try:
                    harness.close()
                except Exception:
                    pass

    def _has_importable_output(self, project: dict) -> bool:
        outputs = Path(studio.PROJECTS_DIR) / project["id"] / "outputs"
        preview = outputs / "preview.html"
        if preview.is_file() and preview.stat().st_size > 80:
            return True
        if project.get("module") == "style-image":
            image = self._find_final_image(outputs)
            return image is not None and image.stat().st_size > 80
        if project.get("module") == "game":
            return any(outputs.glob("*.blend")) and (outputs / "target.png").is_file()
        if project.get("module") == "game-port":
            import port_games

            slug = str(project.get("game_slug") or "").strip()
            web = port_games.PORT_ROOT / "target" / slug / "web" / "index.html"
            return web.is_file() and web.stat().st_size > 40
        if project.get("module") == "adventure-reader":
            book = outputs / "book.json"
            return book.is_file() and book.stat().st_size > 80
        return False

    def _write_run_log(self, project: dict, run_id: str, text: str) -> Path:
        folder = Path(studio.PROJECTS_DIR) / project["id"] / "outputs" / "_logs"
        folder.mkdir(parents=True, exist_ok=True)
        path = folder / f"run-{run_id}.log"
        path.write_text(text, encoding="utf-8")
        return path

    def _record_failure(
        self,
        run_id: str,
        project: dict,
        message: str,
        *,
        finish_reason: Optional[str] = None,
        traceback_text: str = "",
        status: str = "failed",
        label: str = "构建未能继续",
        detail: str = "",
    ) -> None:
        current = self.get(run_id) or {}
        percent = current.get("progress", {}).get("percent", 0)
        log_body = [
            f"run_id={run_id}",
            f"project_id={project.get('id')}",
            f"module={project.get('module')}",
            f"status={status}",
            f"finish_reason={finish_reason or current.get('finish_reason') or ''}",
            f"error={message}",
            "",
            "final_response:",
            str(current.get("final_response") or ""),
        ]
        if traceback_text:
            log_body.extend(["", "traceback:", traceback_text])
        try:
            log_path = self._write_run_log(project, run_id, "\n".join(log_body))
            message = f"{message}\n日志：{log_path}"
        except OSError as exc:
            logger.warning("could not write run log for %s: %s", run_id, exc)
        self._set_progress(run_id, "failed", percent, label, detail or message)
        with self._lock:
            run = self._runs[run_id]
            run["status"] = status
            run["error"] = message[:4000]
            run["updated"] = time.time()
            self._persist(run)
        studio.update_status(project["id"], status, run_id, message[:1000])
        event = {"kind": "error", "text": message}
        if traceback_text:
            event["traceback"] = traceback_text[-4000:]
        self._append(run_id, event)

    def _import_preview(self, project: dict) -> str:
        outputs = Path(studio.PROJECTS_DIR) / project["id"] / "outputs"
        preview = outputs / "preview.html"
        artifact_name = "preview.html"
        if preview.exists():
            document = preview.read_bytes().decode("utf-8", errors="replace")
            document = studio.make_preview_file_safe(document)
            preview.write_text(document, encoding="utf-8")
        elif project.get("module") == "style-image":
            image = self._find_final_image(outputs)
            if image is None:
                return ""
            document = self._image_preview_document(image, project["title"])
            artifact_name = image.name
        else:
            return ""
        share_kwargs = dict(
            title=project["title"],
            owner_id=str(project.get("owner_id") or ""),
            module=str(project.get("module") or ""),
            brief=str(project.get("brief") or "")[:500],
            project_id=str(project.get("id") or ""),
        )
        existing_id = ""
        if project.get("module") == "idea-watch":
            for item in reversed(project.get("outputs") or []):
                if item.get("page_id"):
                    existing_id = item["page_id"]
                    break
        meta = None
        if existing_id:
            meta = storage.update_page(existing_id, document, **share_kwargs)
        if meta is None:
            meta = storage.save_page(document, **share_kwargs)
        studio.add_output(project["id"], artifact_name, meta.id)
        return meta.id

    @staticmethod
    def _find_final_image(outputs: Path) -> Optional[Path]:
        if not outputs.exists():
            return None
        supported = {".png", ".jpg", ".jpeg", ".webp"}
        images = [
            path
            for path in outputs.iterdir()
            if path.is_file() and path.suffix.lower() in supported
        ]
        if not images:
            return None
        final_images = [path for path in images if "final" in path.stem.lower()]
        candidates = final_images or images
        return max(candidates, key=lambda path: path.stat().st_mtime)

    @staticmethod
    def _image_preview_document(image: Path, title: str) -> str:
        mime = mimetypes.guess_type(image.name)[0] or "image/png"
        encoded = base64.b64encode(image.read_bytes()).decode("ascii")
        safe_title = html.escape(title)
        safe_name = html.escape(image.name)
        return f"""<!doctype html>
<html lang="zh-CN"><head><meta charset="utf-8">
<meta name="viewport" content="width=device-width,initial-scale=1">
<title>{safe_title}</title>
<style>
*{{box-sizing:border-box}}body{{margin:0;min-height:100vh;display:grid;place-items:center;
background:#16181f;color:#fff;font-family:system-ui,sans-serif;padding:24px}}
main{{width:min(100%,900px);text-align:center}}h1{{font-size:clamp(20px,4vw,34px);margin:0 0 8px}}
p{{margin:0 0 18px;color:#aeb7c8}}img{{display:block;max-width:100%;max-height:calc(100vh - 150px);
width:auto;height:auto;margin:auto;border-radius:18px;box-shadow:0 18px 60px #0008}}
</style></head><body><main><h1>{safe_title}</h1><p>{safe_name}</p>
<img src="data:{mime};base64,{encoded}" alt="{safe_title}"></main></body></html>"""

    def cancel(self, run_id: str) -> dict:
        with self._lock:
            run = self._runs.get(run_id)
            if run is None:
                raise studio.StudioError("任务不存在")
            if run["status"] not in {"queued", "running"}:
                return json.loads(json.dumps(run))
            run["status"] = "cancel_requested"
            run["updated"] = time.time()
            self._persist(run)
        harness = self._harnesses.get(run_id)
        if harness is not None:
            try:
                harness.close()
            except Exception:
                pass
        return self.get(run_id) or run
